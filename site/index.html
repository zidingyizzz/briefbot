The whole product, one file

This is a real config — a daily tech digest, pushed to Telegram.

config.yaml
# watch → summarize → deliver, described once:
llm:
  base_url: ${OPENAI_BASE_URL:-https://api.openai.com/v1}   # any OpenAI-compatible endpoint
  api_key: ${OPENAI_API_KEY}                            # empty ⇒ plain mode, no key needed
  model: ${OPENAI_MODEL:-gpt-4o-mini}

jobs:
  - name: tech-digest
    subject: "Morning Tech Brief — {today} ({items} items)"
    sources:
      - type: rss
        url: https://techcrunch.com/feed/
        limit: 15
    digest:
      mode: llm          # llm | plain (plain needs no API key)
      only_new: true     # skip items already reported
    channels:
      - type: telegram
        bot_token: ${TELEGRAM_BOT_TOKEN}
        chat_id: ${TELEGRAM_CHAT_ID}
